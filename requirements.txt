# The en_core_web_sm model isn't a normal PyPI package — spaCy ships it as a
# direct-download wheel. Pinning it here (spaCy's officially supported
# approach) means `pip install -r requirements.txt` alone installs it, so a
# deploy step that only runs pip install (e.g. deployment/ansible.yml) can't
# skip it — no separate `spacy download` step to remember or forget.
en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl
